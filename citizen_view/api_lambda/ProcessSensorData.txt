import json
import boto3
import os
from datetime import datetime, timezone

# Initialize AWS clients
dynamodb = boto3.resource('dynamodb')
sns_client = boto3.client('sns')


SNS_TOPIC_ARN = os.environ['SNS_TOPIC_ARN']


def get_pollution_status(temperature, co2, no2):
    # Pollution threshold logic
    if co2 >= 1000 or no2 >= 80 or temperature >= 40:
        return "DANGEROUS"
    elif co2 >= 800 or no2 >= 60 or temperature >= 35:
        return "WARNING"
    return "NORMAL"

def lambda_handler(event, context):
    try:
        # 1. Parse incoming data (Compatible with API Gateway and Test Events)
        if 'body' in event and isinstance(event['body'], str):
            data = json.loads(event['body'])
        elif 'body' in event:
            data = event['body']
        else:
            data = event

        sensor_id = data.get('sensor_id', 'Unknown')
        district = data.get('district', 'Unknown')
        temperature = data.get('temperature', 0)
        co2 = data.get('co2', 0)
        no2 = data.get('no2', 0)
        timestamp = data.get('timestamp', datetime.now(timezone.utc).isoformat())

        # 2. Determine pollution status
        status = get_pollution_status(temperature, co2, no2)

        # 3. Store in DynamoDB
        table = dynamodb.Table('SmartCitySensorData')
        item = {
            'sensor_id': sensor_id,
            'timestamp': timestamp,
            'district': district,
            'temperature': temperature,
            'co2': co2,
            'no2': no2,
            'status': status
        }
        table.put_item(Item=item)
        print(f"Sensor data stored successfully: {item}") # Matching Member 2's log style

        # 4. Trigger SNS Alert
        if status == "DANGEROUS":
            alert_message = f"🚨 EMERGENCY ALERT: Severe pollution detected in SmartCity!\n\n" \
                            f"📍 District: {district} (Sensor: {sensor_id})\n" \
                            f"🌡️ Current Temp: {temperature}°C\n" \
                            f"☁️ Current CO2: {co2} ppm\n" \
                            f"🌫️ Current NO2: {no2} ppb\n\n" \
                            f"Please dispatch an environmental officer for immediate inspection!"
            
            sns_client.publish(
                TopicArn=SNS_TOPIC_ARN,
                Message=alert_message,
                Subject=f"SmartCity Pollution Alert - {district} 🔴"
            )
            print(f"Danger alert sent to email! (Sensor: {sensor_id})")

        # 5. Return API Gateway compatible response
        return {
            "statusCode": 200,
            "headers": {"Content-Type": "application/json"},
            "body": json.dumps({
                "message": "Sensor data processed successfully",
                "data": item
            })
        }

    except Exception as e:
        print("Error:", str(e))
        return {
            "statusCode": 500,
            "body": json.dumps({
                "message": "Internal server error",
                "error": str(e)
            })
        }
